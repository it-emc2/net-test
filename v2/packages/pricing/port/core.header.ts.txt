/** Mongoose-like Product model: only these two query shapes are used. */
export interface PricingProductModel {
  find(q: { productId: { $in: string[] } }): { lean(): Promise<any[]> };
  findOne(q: { productId: string }): { lean(): Promise<any | null> };
}
/** Offer/Draft model as used for snapshot lookups (optional, server only). */
export interface PricingSnapshotModel {
  findOne(q: Record<string, unknown>): any;
}
export interface PricingDeps {
  /** Admin-tunable business numbers. */
  cfg: { get(key: string, fallback?: unknown): any };
  /** articleNumbers → Map<id, netPrice>; an empty Map is the documented fallback. */
  fetchVigourNetPrices: (ids: string[]) => Promise<Map<string, number>>;
  OfferModel?: PricingSnapshotModel;
  DraftModel?: PricingSnapshotModel;
}
